cmake_minimum_required(VERSION 3.25...4.2)
project(IfTruthy LANGUAGES CXX)

set(VALUES ON OFF TRUE FALSE YES NO Y N IGNORE NOTFOUND 1 0 2 -1 0.0 abc foo-NOTFOUND "")
foreach(v IN LISTS VALUES)
  set(X "${v}")
  if(X)
    set(by_name TRUE)
  else()
    set(by_name FALSE)
  endif()
  if(${X})
    set(by_value TRUE)
  else()
    set(by_value FALSE)
  endif()
  message(STATUS "X=[${v}]  if(X)=${by_name}  if(\${X})=${by_value}")
endforeach()

set(features "core;extra")
add_executable(app main.cpp)
if(TARGET app AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp"
   AND "extra" IN_LIST features AND DEFINED CMAKE_CXX_COMPILER)
  target_compile_definitions(app PRIVATE HAVE_EXTRA=1)
endif()
if("${CMAKE_CXX_COMPILER_ID}" STREQUAL "GNU"
   AND CMAKE_CXX_COMPILER_VERSION VERSION_GREATER_EQUAL 15)
  target_compile_definitions(app PRIVATE GCC15=1)
endif()
if(2 EQUAL 2)
  target_compile_definitions(app PRIVATE TWO=1)
endif()
