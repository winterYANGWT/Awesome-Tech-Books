cmake_minimum_required(VERSION 3.25...4.2)
project(ParseArgs LANGUAGES CXX)

add_library(core STATIC core.cpp)
target_include_directories(core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")

function(mylib_add_tool name)
  cmake_parse_arguments(PARSE_ARGV 1 arg "INSTALL" "OUTPUT_NAME" "SOURCES;LIBS")
  message(STATUS "SOURCES=${arg_SOURCES}")
  message(STATUS "LIBS=${arg_LIBS}  INSTALL=${arg_INSTALL}")
  message(STATUS "OUTPUT_NAME=${arg_OUTPUT_NAME}  UNPARSED=${arg_UNPARSED_ARGUMENTS}  MISSING=${arg_KEYWORDS_MISSING_VALUES}")
  if(DEFINED arg_UNPARSED_ARGUMENTS OR DEFINED arg_KEYWORDS_MISSING_VALUES)
    message(FATAL_ERROR "mylib_add_tool(${name})：不认识的参数 '${arg_UNPARSED_ARGUMENTS}'，缺值的参数 '${arg_KEYWORDS_MISSING_VALUES}'")
  else()
    add_executable(${name} ${arg_SOURCES})
    target_link_libraries(${name} PRIVATE ${arg_LIBS})
    set_target_properties(${name} PROPERTIES OUTPUT_NAME "${arg_OUTPUT_NAME}")
    if(arg_INSTALL)
      install(TARGETS ${name} RUNTIME DESTINATION bin)
    endif()
    message(STATUS "已创建 ${arg_OUTPUT_NAME}，链接 ${arg_LIBS}")
  endif()
endfunction()

  mylib_add_tool(greet SOURCE main.cpp OUTPUT_NAME)
